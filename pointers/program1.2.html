Program 02 - Pointer and 2D-Array

Problem Statement

Write a program to input and display the elements of a 2D array of size 2 x 3 using a pointer.

Source Code

#include <stdio.h>

void main()
{
	int a[2][3], i, j;
	int *p;

	p = a;

	for(i = 1; i <= 6; i++)
	{
		printf("Enter %d value: ", i);
		scanf("%d", p);
		p++;
	}

	p = a;

	printf("\nArray values: \n");
	for(i = 0; i < 2; i++)
	{
		for(j = 0; j < 3; j++)
		{
			printf("%d ",*p);
			p++;
		}
		printf("\n");
	}
}
