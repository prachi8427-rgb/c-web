Program 06 - Count The Char in File

Problem Statement

Create a file and count the no. of char in it.

Source Code

#include <stdio.h>
#include <conio.h>

void main()
{
    char f[10], data[30], a;
    int x, count = 0;
    FILE * New;
	printf("Enter file name: ");
	gets(f);
	printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");
	New = fopen(f,"w");
	do{
        gets(data);
        fprintf(New, "%s\n", data);
        a = getch();}
        while(a != 19);
        printf("\nSaved successfully\n\n");
    fclose(New);

    New = fopen(f,"r");
    do{
    	x = fgetc(New);
    	if(x != -1)
    	count++;
    }while(x != -1);
    fclose(New);

    printf("File have total = %d characters.",count);
}
