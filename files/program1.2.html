Program 02 - Save The File When User Want

Problem Statement

Enter file name and data from user, data enter till user want to enter the data.

Source Code

#include <stdio.h>
#include <conio.h>

void main()
{
    char f[10], data[30], a;
	FILE * New;
	printf("Enter file name: ");
	gets(f);
	printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");
	New = fopen(f,"w");
	do{
        gets(data);
        fprintf(New, "%s\n", data);
        a = getch();}
        while(a != 19);
        printf("\nSaved successfully");
    fclose(New);
}
