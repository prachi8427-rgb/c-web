Program 07 - Count The Words in File

Problem Statement

Create a file and count the no. of words in it.

Source Code

#include <stdio.h>
#include <conio.h>

void main()
{
    char f[10], data[50], a, *c;
    int x, count = 1;
    FILE * New;
	printf("Enter file name: ");
	gets(f);
	printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");
	New = fopen(f,"w");
	do{
        gets(data);
        fprintf(New, "%s\n", data);
        a = getch();}
        while(a != 19);
        printf("\nSaved successfully\n\n");
    fclose(New);

    
    New = fopen(f, "r");
    do{
    	c = fgets(data, 50, New);
    	if(c != 0){
    		for(x = 0; data[x] != '\0'; x++){
    			if(data[x] == ',' || data[x] == ' '){
    				count++;
    				x++;
    			}
    		}
    	}
    	
    }while(c != 0);
    fclose(New);

    printf("File have total = %d words.",count);
}
