Program 03 - Copy File

Problem Statement

Create a file and copy that file into another file.

Source Code

#include <stdio.h>
#include <conio.h>

void main()
{
    char f[10], data[30], a, *c;
	FILE * New, *New2;
	printf("Enter file name: ");
	gets(f);
	printf("First enter then use CTRL + S to save... \nFor next line press enter 2 times...\nEnter the data: ");
	New = fopen(f,"w");
	do{
        gets(data);
        fprintf(New, "%s\n", data);
        a = getch();}
        while(a != 19);
        printf("\nSaved successfully\n");
    fclose(New);

    New = fopen(f,"r");
    printf("Enter 2nd file name: ");
    gets(f);
    New2 =  fopen(f,"w");
    do{
    	c = fgets(data, 50, New);
    	if(c != 0)
    	  fprintf(New2, "%s", data);
    }while(c != 0);
    fclose(New2);
    fclose(New);
    printf("\nData copied successfully.");
}
