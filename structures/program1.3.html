Program 03 - Employees Name Characters with ID

Problem Statement

Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form.

Source Code

#include <stdio.h>

struct emp
{
	int id;
	char name[20];
	char designation[30];
};

void main()
{
	struct emp e[50];
	int i, j;

	for(i = 0; i < 50; i++)
	{
		printf("\nEnter details of employee %d\n", i + 1);

		printf("Enter Employee Id: ");
		scanf("%d",&e[i].id);

		printf("Enter Name: ");
		scanf("%s",e[i].name);

		printf("Enter Designation: ");
		scanf("%s",e[i].designation);
	}

	printf("\n\nEMPLOYEE DETAILS\n");
	printf("------------------\n");
	printf("ID\tNAME CHARACTERS\n");
	printf("------------------\n");

	for(i = 0; i < 50; i++)
	{
	  printf("%d\t", e[i].id);
		for(j = 0; e[i].name[j] != '\0'; j++)
		{
		  printf("%c ",e[i].name[j]);
		}
	  printf("\n");
	}
}
